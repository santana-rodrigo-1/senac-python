1) Custo de Viagem

distancia = float(input("Distância da viagem (km): "))
consumo = float(input("Consumo do veículo (km/l): "))
preco = float(input("Preço da gasolina: R$ "))

litros = distancia / consumo
custo = litros * preco

print("\n===== RESUMO =====")
print(f"Litros necessários: {litros:.2f} L")
print(f"Custo da viagem: R$ {custo:.2f}")
