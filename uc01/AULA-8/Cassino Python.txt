import random

def cara_ou_coroa():
    print(random.choice(["Cara", "Coroa"]))


import random

def lancar_dado():
    print(random.randint(1, 6))


import random

def numero_secreto():

    numero = random.randint(1, 20)

    while True:

        chute = int(input("Número: "))

        if chute < numero:
            print("Maior")

        elif chute > numero:
            print("Menor")

        else:
            print("Acertou!")
            break


import random

def mega_sena():

    numeros = random.sample(range(1, 61), 6)

    print(numeros)


from datetime import datetime

def data_hora():

    print(datetime.now())



while True:

    print("\n===== CASSINO PYTHON =====")
    print("1 - Cara ou Coroa")
    print("2 - Lançar Dado")
    print("3 - Número Secreto")
    print("4 - Mega Sena")
    print("5 - Data e Hora")
    print("6 - Sair")

    op = input("Escolha: ")

    if op == "1":
        cara_ou_coroa()

    elif op == "2":
        lancar_dado()

    elif op == "3":
        numero_secreto()

    elif op == "4":
        mega_sena()

    elif op == "5":
        data_hora()

    elif op == "6":
        break

    else:
        print("Opção inválida!")
