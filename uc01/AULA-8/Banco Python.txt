def depositar(saldo, valor):
    return saldo + valor


def sacar(saldo, valor):
    if valor > saldo:
        print("Saldo insuficiente!")
        return saldo
    return saldo - valor


def extrato(saldo):
    print(f"Saldo atual: R$ {saldo:.2f}")


saldo = 0

while True:

    print("\n===== BANCO PYTHON =====")
    print("1 - Depositar")
    print("2 - Sacar")
    print("3 - Extrato")
    print("4 - Sair")

    op = input("Escolha: ")

    if op == "1":
        valor = float(input("Valor: "))
        saldo = depositar(saldo, valor)

    elif op == "2":
        valor = float(input("Valor: "))
        saldo = sacar(saldo, valor)

    elif op == "3":
        extrato(saldo)

    elif op == "4":
        break

    else:
        print("Opção inválida!")
