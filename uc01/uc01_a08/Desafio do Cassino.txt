import random
from datetime import datetime

saldo = 100
partidas = 0
vitorias = 0


def cara_ou_coroa():
    return random.choice(["Cara", "Coroa"])


def dado():
    return random.randint(1, 6)


def numero_secreto():

    numero = random.randint(1, 10)

    while True:

        chute = int(input("Número: "))

        if chute == numero:
            return True

        elif chute < numero:
            print("Maior")

        else:
            print("Menor")


def mega():

    print(random.sample(range(1, 61), 6))
    return False


while saldo > 0:

    print(f"\nSaldo: R$ {saldo}")

    print("""
1 - Cara ou Coroa
2 - Dado
3 - Número Secreto
4 - Mega Sena
5 - Ranking
6 - Sair
""")

    op = input("Escolha: ")

    if op == "6":
        break

    if op == "5":
        print(f"Partidas: {partidas}")
        print(f"Vitórias: {vitorias}")
        continue

    saldo -= 10
    partidas += 1

    print("Data:", datetime.now().strftime("%d/%m/%Y %H:%M"))

    venceu = False

    if op == "1":

        escolha = input("Cara ou Coroa: ").capitalize()

        if escolha == cara_ou_coroa():
            venceu = True

    elif op == "2":

        numero = int(input("Escolha um número de 1 a 6: "))

        if numero == dado():
            venceu = True

    elif op == "3":

        venceu = numero_secreto()

    elif op == "4":

        mega()

    if venceu:
        saldo += 20
        vitorias += 1
        print("Você venceu! +R$20")
    else:
        print("Você perdeu!")

print("\n===== RANKING =====")
print(f"Partidas jogadas: {partidas}")
print(f"Vitórias: {vitorias}")
print(f"Saldo final: R$ {saldo:.2f}")
