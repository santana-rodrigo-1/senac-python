renda = float(input("Informe sua renda: R$ "))
score = int(input("Informe seu score: "))

if renda >= 3000 and score >= 700:
    print("Empréstimo aprovado!")
else:
    print("Empréstimo negado.")

media = float(input("Informe sua média: "))
renda_familiar = float(input("Informe a renda familiar: R$ "))

if media >= 9 or renda_familiar <= 2000:
    print("Você tem direito à bolsa de estudos!")
else:
    print("Você não possui os requisitos para a bolsa.")

inadimplente = input("O cliente está inadimplente? (s/n): ")

if not inadimplente == "s":
    print("Compra liberada!")
else:
    print("Compra bloqueada.")
