print("1 - Carro Popular")
print("2 - Sedan")
print("3 - SUV")
print("4 - Cancelar")

opcao = input("Escolha um veículo: ")

match opcao:

    case "1":
        dias = int(input("Quantidade de dias: "))

        if dias >= 1:
            print("Carro Popular reservado!")
        else:
            print("Quantidade de dias inválida.")

    case "2":
        dias = int(input("Quantidade de dias: "))

        if dias >= 1:
            print("Sedan reservado!")
        else:
            print("Quantidade de dias inválida.")

    case "3":
        dias = int(input("Quantidade de dias: "))

        if dias >= 1:
            print("SUV reservado!")
        else:
            print("Quantidade de dias inválida.")

    case "4":
        print("Reserva cancelada.")

    case _:
        print("Opção inválida.")

        if dias >= 1:

            if opcao == "1":

            else:
                print("SUV reservado com sucesso!")

        else:
            print("Quantidade de dias inválida.")

    case "4":

    case _:
        print("Opção inválida.")
