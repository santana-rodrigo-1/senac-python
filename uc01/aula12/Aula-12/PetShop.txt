class Pet:
    def __init__(self, nome, especie, preco_banho):
        self.nome = nome
        self.especie = especie
        self.preco_banho = preco_banho
        self.__atendimentos = 0  # Atributo privado

    def dar_banho(self):
        self.__atendimentos += 1
        print(f"🧼 O pet {self.nome} tomou um banho!")

    def consultar_atendimentos(self):
        return self.__atendimentos

    def calcular_total(self):
        return self.preco_banho * self.__atendimentos


class PetVIP(Pet):
    def __init__(self, nome, especie, preco_banho, desconto_vip=10.0):
        # Inicialização explícita chamando a classe mãe Pet.__init__ 
        Pet.__init__(self, nome, especie, preco_banho)
        self.desconto_vip = desconto_vip

    # Polimorfismo: Sobrescrita do cálculo usando apenas self
    def calcular_total(self):
        subtotal = self.preco_banho * self.consultar_atendimentos()
        total_com_desconto = subtotal - self.desconto_vip
        if total_com_desconto < 0:
            return 0.0
        return total_com_desconto


# REPOSITÓRIO EM MEMÓRIA E CRUD
pets = []

def buscar_pet(nome):
    for p in pets:
        if p.nome.lower() == nome.lower():
            return p
    return None

def cadastrar_pet():
    print("\n--- CADASTRO DE PET ---")
    nome = input("Nome do Pet: ")
    if buscar_pet(nome) != None:
        print("❌ Pet com este nome já cadastrado!")
        return

    especie = input("Espécie (Cão, Gato, etc): ")
    preco = float(input("Preço do Banho: R$ "))
    tipo = input("Tipo de Cliente (1-Comum, 2-VIP): ")

    if tipo == "1":
        novo_pet = Pet(nome, especie, preco)
    elif tipo == "2":
        novo_pet = PetVIP(nome, especie, preco)
    else:
        print("❌ Tipo inválido!")
        return

    pets.append(novo_pet)
    print("✅ Pet cadastrado com sucesso!")

def listar_pets():
    if len(pets) == 0:
        print("ℹ️ Nenhum pet cadastrado.")
    else:
        print("\n---------------- LISTA DE PETS ----------------")
        for p in pets:
            tipo_str = "VIP" if isinstance(p, PetVIP) else "Comum"
            print(f"Nome: {p.nome} | Espécie: {p.especie} | Tipo: {tipo_str} | Banhos: {p.consultar_atendimentos()} | Total: R$ {p.calcular_total():.2f}")
        print("-----------------------------------------------")

def dar_banho_pet():
    nome = input("Digite o nome do pet que vai tomar banho: ")
    p = buscar_pet(nome)
    if p != None:
        p.dar_banho()
    else:
        print("❌ Pet não encontrado!")


# MENU PRINCIPAL
while True:
    print("\n=== PETSHOP AMIGOPET ===")
    print("1 - Cadastrar Pet")
    print("2 - Listar Pets")
    print("3 - Registrar Banho")
    print("0 - Sair")

    op = input("Opção: ")
    if op == "1": cadastrar_pet()
    elif op == "2": listar_pets()
    elif op == "3": dar_banho_pet()
    elif op == "0":
        print("Encerrando o PetShop... Até logo!")
        break
    else:
        print("Opção inválida!")
