saldo = 1000

print("===== SISTEMA JWC =====")
print("1 - Caixa Eletrônico")
print("2 - Empréstimo Bancário")
print("3 - Bolsa de Estudos")
print("4 - Pedra, Papel e Tesoura")
print("5 - Sair")

opcao = input("Escolha uma opção: ")

match opcao:

    # Caixa Eletrônico
    case "1":

        print("\n=== CAIXA ELETRÔNICO ===")
        print("1 - Saque")
        print("2 - Depósito")
        print("3 - Consultar Saldo")

        operacao = input("Escolha uma operação: ")

        match operacao:

            case "1":
                valor = float(input("Valor do saque: R$ "))

                if valor > 0 and valor <= saldo:
                    saldo = saldo - valor
                    print("Saque realizado com sucesso!")
                    print(f"Saldo atual: R$ {saldo:.2f}")
                else:
                    print("Saque não autorizado.")

            case "2":
                valor = float(input("Valor do depósito: R$ "))

                if valor > 0:
                    saldo = saldo + valor
                    print("Depósito realizado com sucesso!")
                    print(f"Saldo atual: R$ {saldo:.2f}")
                else:
                    print("Valor inválido.")

            case "3":
                print(f"Saldo disponível: R$ {saldo:.2f}")

            case _:
                print("Operação inválida.")

    # Empréstimo
    case "2":

        print("\n=== EMPRÉSTIMO BANCÁRIO ===")

        renda = float(input("Informe sua renda: R$ "))
        score = int(input("Informe seu score: "))

        if renda >= 3000 and score >= 700:
            print("Empréstimo aprovado!")
        else:
            print("Empréstimo negado.")

    # Bolsa
    case "3":

        print("\n=== BOLSA DE ESTUDOS ===")

        media = float(input("Informe sua média: "))
        renda = float(input("Informe sua renda familiar: R$ "))

        if media >= 9 or renda <= 2000:
            print("Bolsa concedida!")
        else:
            print("Bolsa não concedida.")

    # Pedra, Papel e Tesoura
    case "4":

        print("\n=== PEDRA, PAPEL E TESOURA ===")

        jogador1 = input("Jogador 1 (pedra, papel ou tesoura): ")
        jogador2 = input("Jogador 2 (pedra, papel ou tesoura): ")

        match jogador1:

            case "pedra":

                if jogador2 == "pedra":
                    print("Empate!")
                elif jogador2 == "papel":
                    print("Jogador 2 venceu!")
                elif jogador2 == "tesoura":
                    print("Jogador 1 venceu!")
                else:
                    print("Jogada inválida do Jogador 2.")

            case "papel":

                if jogador2 == "papel":
                    print("Empate!")
                elif jogador2 == "pedra":
                    print("Jogador 1 venceu!")
                elif jogador2 == "tesoura":
                    print("Jogador 2 venceu!")
                else:
                    print("Jogada inválida do Jogador 2.")

            case "tesoura":

                if jogador2 == "tesoura":
                    print("Empate!")
                elif jogador2 == "papel":
                    print("Jogador 1 venceu!")
                elif jogador2 == "pedra":
                    print("Jogador 2 venceu!")
                else:
                    print("Jogada inválida do Jogador 2.")

            case _:
                print("Jogada inválida do Jogador 1.")

    # Sair
    case "5":
        print("Programa encerrado.")

    # Opção inválida
    case _:
        print("Opção inválida.")
