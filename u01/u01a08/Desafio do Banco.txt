from datetime import datetime

historico = []
saldo = 0
saques = 0
total_depositado = 0
total_sacado = 0
maior_deposito = 0


def depositar(saldo, valor):
    global total_depositado, maior_deposito

    if valor <= 0:
        print("Valor inválido!")
        return saldo

    saldo += valor
    total_depositado += valor

    if valor > maior_deposito:
        maior_deposito = valor

    historico.append(
        f"{datetime.now().strftime('%d/%m/%Y %H:%M')} - Depósito: R$ {valor:.2f}"
    )

    return saldo


def sacar(saldo, valor):
    global saques, total_sacado

    if saques >= 3:
        print("Limite diário de saques atingido!")
        return saldo

    if valor > saldo:
        print("Saldo insuficiente!")
        return saldo

    saldo -= valor
    saques += 1
    total_sacado += valor

    historico.append(
        f"{datetime.now().strftime('%d/%m/%Y %H:%M')} - Saque: R$ {valor:.2f}"
    )

    return saldo


def extrato(saldo):
    print(f"\nSaldo: R$ {saldo:.2f}")


def mostrar_historico():
    print("\n=== HISTÓRICO ===")

    if len(historico) == 0:
        print("Nenhuma operação.")
    else:
        for item in historico:
            print(item)


def limpar_historico():
    historico.clear()
    print("Histórico apagado!")


while True:

    print("""
1 - Depositar
2 - Sacar
3 - Extrato
4 - Histórico
5 - Limpar Histórico
6 - Resumo
7 - Sair
""")

    op = input("Escolha: ")

    if op == "1":
        saldo = depositar(saldo, float(input("Valor: ")))

    elif op == "2":
        saldo = sacar(saldo, float(input("Valor: ")))

    elif op == "3":
        extrato(saldo)

    elif op == "4":
        mostrar_historico()

    elif op == "5":
        limpar_historico()

    elif op == "6":
        print(f"Total depositado: R$ {total_depositado:.2f}")
        print(f"Total sacado: R$ {total_sacado:.2f}")
        print(f"Maior depósito: R$ {maior_deposito:.2f}")

    elif op == "7":
        break
